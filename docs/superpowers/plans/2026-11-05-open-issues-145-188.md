import { type BucketAgg, mergeAgg } from '../../src/rollups/metric-agg.js';

/** The one bucket (series 0, index 7) a fresh buffer holds after `aggs` arrive in this order. */
function buffered(...aggs: BucketAgg[]): BucketAgg | undefined {
  const b = new MetricBuffer(2);
  for (const a of aggs) b.add(0, 7, a, 'p');
  const s = b.series('p');
  expect(s.map((x) => x.t)).toEqual([[7]]);
  return s[0]?.a[0];

describe('MetricBuffer', () => {
  test('equal-time gauge points keep the larger last either way round, as mergeAgg does (#147)', () => {
    const want = { k: 'number', count: 2, sum: 8, min: 3, max: 5, last: 5, lastTime: 1 };
    expect(mergeAgg(num(5, 1), num(3, 1), 'p')).toEqual(want);
    expect(mergeAgg(num(3, 1), num(5, 1), 'p')).toEqual(want);
    expect(buffered(num(5, 1), num(3, 1))).toEqual(want);
    expect(buffered(num(3, 1), num(5, 1))).toEqual(want);

**Files:**

**Interfaces:**

  `compileRegex` already builds (`'u'` plus `i`/`s` from a leading `(?i)`/`(?s)`).
- The message (exact; in TS source the backslashes are doubled):
  `nested repetition (e.g. "(a+)+" or "(a+){2}") can backtrack catastrophically; put a literal the repeated part cannot match after it, like (\d+\.){3}, or repeat a class instead, like [a-z]{1,64}`
- The backreference and lookaround checks and their messages are unchanged.

**The rule** (exact; the tests do not determine every branch):
- A *variable quantifier* is `*`, `+`, `{n,}`, or `{n,m}` with `m > n` and `m > 1` (`?` and `{n}` are not). A trailing
  lazy `?` belongs to the quantifier before it.
- A quantifier with maximum above 1 on a **group** G that contains a variable quantifier anywhere inside it
  (descending into subgroups) is rejected unless, for **every** variable quantifier X inside G, the follower of X is
  safe.
- The *follower* of X is the item right after X's quantifier in X's sequence. If X is the last item of a subgroup
  (inside G), the follower is the item after that subgroup's `)` (repeat outward); reaching G's own `)` means no
  follower. `|`, `^`, `$`, a group start, or the end of G means no safe follower.
- The follower is safe iff (a) it is a single literal: one plain code point other than `^$\.*+?()[]{}|`, or `\`
  followed by one of `^$\.*+?()[]{}|/`, never a class, a class escape (`\d \D \w \W \s \S`), another escape, `.` or a
  group; (b) it is mandatory: not followed by a quantifier with minimum 0 (`?`, `*`, `{0,…}`); and (c)
  `new RegExp('^(?:' + atomText + ')$', flags).test(char)` is false, where `atomText` is the source text of X's
  quantified atom (a char, escape, class, `.`, or a whole group including its parens). If that `new RegExp` throws,
  treat the follower as unsafe.
- Builder's pick for the parser: one pass building a small tree of items (atom / group with children / `|` /
  anchor), each with its source span and optional `{ min, max }`; then the check above per quantified group. Pattern
  length stays capped at 512, so a quadratic walk is fine.
