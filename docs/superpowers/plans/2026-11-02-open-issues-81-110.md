  pull_request) base="$BASE_SHA"; head="$HEAD_SHA" ;;
  push) base="$BEFORE_SHA"; head="$GITHUB_SHA" ;;

docs_only=true
while IFS= read -r f; do

[ -n "$files" ] || docs_only=false
echo "docs_only=$docs_only" >> "$GITHUB_OUTPUT"
